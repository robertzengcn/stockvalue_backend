   Implementation pattern (follow existing `get_profit_sheet` style):
   ```python
   async def fetch_multi_year_financials(
       self,
       ts_code: str,
       years: int = 3,
   ) -> list[dict[str, Any]]:
       """Fetch multi-year financial data (profit + balance sheet) per D-04.

       Calls AKShare once (returns all years), filters in-memory for the
       requested number of years.
       """
       def _fetch() -> list[dict[str, Any]]:
           import akshare as ak
           import pandas as pd

           em = eastmoney_hsf10_symbol(ts_code)
           # Get ALL profit sheet periods
           profit_df = ak.stock_profit_sheet_by_report_em(symbol=em)
           # Get ALL balance sheet periods
           balance_df = ak.stock_balance_sheet_by_report_em(symbol=em)
           # ... filter, merge, return list of dicts
       return await self._run_sync(_fetch)
   ```
