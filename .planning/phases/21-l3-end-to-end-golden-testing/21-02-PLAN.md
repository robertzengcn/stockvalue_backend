```python
"""L3 end-to-end golden pipeline tests using frozen AKShare data.

Full pipeline: frozen JSON -> build report -> call services -> compare against expected values.
No network required. Parametrized over all l3_verified stocks in manifest.yaml.
"""
from __future__ import annotations

import math
from typing import Any

import pytest

from stockvaluefinder.validation.comparators import ComparisonResult


P0_METRICS = [
    "dsri", "gmi", "aqi", "sgi", "depi", "sgai", "lvgi", "tata",
    "m_score", "f_score", "nopat", "invested_capital", "roic",
]

P1_METRICS = [
    "detect_存贷双高", "goodwill_ratio", "profit_cash_divergence",
]

# Metrics that require external data (always null in expected_metrics.yaml)
SKIP_METRICS = [
    "roic_wacc_spread", "wacc", "present_value", "terminal_value",
    "margin_of_safety", "net_dividend_yield", "yield_gap",
    "buyback_yield", "capital_allocation_score",
    "resonance_score", "dcf_adjustment", "alpha_score",
]
```

**Test class: `TestL3GoldenFrozen`** with `@pytest.mark.golden` marker.

**Parametrize helper:** Define a function `_get_verified_ids` that reads manifest.yaml and returns only (ticker, year) tuples where l3_verified=true. Use this in a `pytest.parametrize` decorator on each test method, with `ids` formatted as `"{ticker}_{year}"`.

Actually, since `verified_golden_stock_ids` is a session-scoped fixture, use indirect parametrization. Define a helper that reads manifest directly:

```python
def _load_verified_ids():
    """Load verified (ticker, year) pairs from manifest for parametrize."""
    from pathlib import Path
    import yaml
    manifest_path = Path(__file__).parent / "manifest.yaml"
    manifest = yaml.safe_load(manifest_path.read_text(encoding="utf-8"))
    ids = []
    for entry in manifest["golden_stocks"]:
        if entry.get("l3_verified", False):
            for year in entry["years"]:
                ids.append((entry["ticker"], year, entry.get("is_financial", False)))
    return ids
```

**Test method: `test_l3_p0_metrics`** -- For each verified stock, load expected metrics via `golden_loader`, compute metrics via `compute_metrics_from_frozen`, then iterate over P0_METRICS. For each metric:
- If expected value is None (skip_reason present), skip with `pytest.skip(f"Metric {name} has no expected value: {skip_reason}")`
- If computed value is None (previous year data missing), skip with `pytest.skip(f"Could not compute {name}: previous year data unavailable")`
- Otherwise, convert bool expected values to float (True->1.0, False->0.0) and call `assert_metric_within_tolerance(name, expected_float, computed_float, registry)`
- P0 failures are hard failures (plain assert, no xfail)

**Test method: `test_l3_p1_metrics`** -- Same structure but for P1_METRICS. P1 metric failures should use `pytest.xfail` with reason rather than hard failure. Wrap the assertion in a try/except AssertionError, and if it fails, call `pytest.xfail(f"P1 metric {name} outside tolerance: expected={expected}, computed={computed}")`. This means the test still runs and reports the failure, but does not break CI.

**Test method: `test_l3_all_metrics_counted`** -- Verify that every metric in expected_metrics.yaml is either in P0_METRICS, P1_METRICS, or SKIP_METRICS. This catches cases where new metrics are added to expected_metrics.yaml but not categorized. This is a non-parametrized test.

```python
@pytest.mark.golden
class TestL3GoldenFrozen:

    @pytest.mark.parametrize(
        "ticker,year,is_financial",
        _load_verified_ids(),
        ids=[f"{t}_{y}" for t, y, _ in _load_verified_ids()],
    )
    def test_l3_p0_metrics(
        self,
        ticker: str,
        year: int,
        is_financial: bool,
        golden_loader,
        compute_metrics_from_frozen,
        metric_registry_fixture,
        assert_metric_within_tolerance,
    ):
        expected_data = golden_loader(ticker, year)
        computed = compute_metrics_from_frozen(ticker, year, is_financial)
        metrics = expected_data["metrics"]

        failures = []
        for name in P0_METRICS:
            if name not in metrics:
                continue
            entry = metrics[name]
            expected = entry.get("value")
            if expected is None:
                skip_reason = entry.get("skip_reason", "no expected value")
                pytest.skip(f"Metric {name}: {skip_reason}")
                return  # skip the rest for this stock
            computed_val = computed.get(name)
            if computed_val is None:
                pytest.skip(f"Could not compute {name}")
                return
            # Convert bool to float for comparison
            if isinstance(expected, bool):
                expected = 1.0 if expected else 0.0
            if isinstance(computed_val, bool):
                computed_val = 1.0 if computed_val else 0.0
            try:
                assert_metric_within_tolerance(
                    name, float(expected), float(computed_val), metric_registry_fixture
                )
            except AssertionError as e:
                failures.append(str(e))

        assert not failures, (
            f"P0 metric failures for {ticker}/{year}:\n" +
            "\n".join(failures)
        )

    @pytest.mark.parametrize(
        "ticker,year,is_financial",
        _load_verified_ids(),
        ids=[f"{t}_{y}" for t, y, _ in _load_verified_ids()],
    )
    def test_l3_p1_metrics(
        self,
        ticker: str,
        year: int,
        is_financial: bool,
        golden_loader,
        compute_metrics_from_frozen,
        metric_registry_fixture,
        assert_metric_within_tolerance,
    ):
        expected_data = golden_loader(ticker, year)
        computed = compute_metrics_from_frozen(ticker, year, is_financial)
        metrics = expected_data["metrics"]

        for name in P1_METRICS:
            if name not in metrics:
                continue
            entry = metrics[name]
            expected = entry.get("value")
            if expected is None:
                continue  # P1: silently skip null expected values
            computed_val = computed.get(name)
            if computed_val is None:
                continue  # P1: silently skip if uncomputable
            if isinstance(expected, bool):
                expected = 1.0 if expected else 0.0
            if isinstance(computed_val, bool):
                computed_val = 1.0 if computed_val else 0.0
            try:
                assert_metric_within_tolerance(
                    name, float(expected), float(computed_val), metric_registry_fixture
                )
            except AssertionError:
                pytest.xfail(
                    f"P1 metric '{name}' outside tolerance for {ticker}/{year}"
                )

    def test_l3_all_metrics_counted(self, golden_loader, verified_golden_stock_ids):
        """Verify every metric in expected_metrics.yaml is categorized."""
        all_known = set(P0_METRICS) | set(P1_METRICS) | set(SKIP_METRICS)
        for ticker, year in verified_golden_stock_ids:
            expected_data = golden_loader(ticker, year)
            for metric_name in expected_data.get("metrics", {}):
                assert metric_name in all_known, (
                    f"Metric '{metric_name}' in {ticker}/{year} expected_metrics.yaml "
                    f"is not categorized as P0, P1, or SKIP"
                )
```

```python
"""L3 live golden tests against real AKShare endpoints.

Runs weekly (not on every PR) to detect upstream data changes or field renames.
Requires network access and DEVELOPMENT_MODE=true.
"""
from __future__ import annotations

import pytest

from stockvaluefinder.external.data_service import ExternalDataService


SKIP_REASON = (
    "Live golden tests require network access. "
    "Run with: pytest -m golden_live"
)


def _load_verified_ids():
    """Load verified (ticker, year) pairs from manifest for parametrize."""
    from pathlib import Path
    import yaml
    manifest_path = Path(__file__).parent / "manifest.yaml"
    manifest = yaml.safe_load(manifest_path.read_text(encoding="utf-8"))
    ids = []
    for entry in manifest["golden_stocks"]:
        if entry.get("l3_verified", False):
            for year in entry["years"]:
                ids.append((entry["ticker"], year, entry.get("is_financial", False)))
    return ids


@pytest.mark.golden_live
class TestL3GoldenLive:
    """Live AKShare golden tests. Run weekly to detect upstream data changes."""

    @pytest.mark.parametrize(
        "ticker,year,is_financial",
        _load_verified_ids(),
        ids=[f"{t}_{y}" for t, y, _ in _load_verified_ids()],
    )
    @pytest.mark.asyncio
    async def test_l3_live_data_fetch(
        self,
        ticker: str,
        year: int,
        is_financial: bool,
    ):
        """Verify AKShare can fetch data for this ticker/year.

        This test does NOT compare values -- it only verifies that the
        AKShare API returns data for the golden stock. Value comparison
        is done by the frozen L3 tests after manual data refresh.
        """
        service = ExternalDataService(
            tushare_token="",
            enable_akshare=True,
        )
        await service.initialize()

        report = await service.get_financial_report(ticker, year)

        assert report is not None, f"AKShare returned None for {ticker}/{year}"
        assert report.get("revenue") is not None, (
            f"AKShare report for {ticker}/{year} has no revenue field -- "
            f"possible upstream field rename"
        )
        assert float(report.get("revenue", 0)) != 0.0, (
            f"AKShare report for {ticker}/{year} has zero revenue"
        )
```

The live test only verifies data is fetchable and revenue is non-zero. It does NOT do value comparison because live data may differ from frozen data. Its purpose is to detect API breakage (field renames, endpoint changes), not value drift.
