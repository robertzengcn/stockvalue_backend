The diff report format is a list of dicts:
```python
[
    {
        "metric_name": "m_score",
        "expected": -0.7421,
        "computed": -0.74,
        "delta": 0.0021,
        "tolerance": {"absolute": 0.05},
        "passed": True,
    },
    ...
]
```

Implement these helper functions at module level (they are test utilities, not production code):

1. **`generate_diff_report(comparisons: list[ComparisonResult]) -> list[dict[str, Any]]`**
   Takes a list of ComparisonResult objects and returns the structured diff format. Each entry includes metric_name, expected, computed, delta, tolerance (as dict with absolute/relative keys, None values excluded), and passed boolean.

```python
def generate_diff_report(comparisons: list[ComparisonResult]) -> list[dict[str, Any]]:
    """Convert ComparisonResult list to structured diff report.

    Each entry: {metric_name, expected, computed, delta, tolerance, passed}
    Tolerance is serialized as dict with only non-None keys.
    """
    report = []
    for cmp in comparisons:
        tol_dict: dict[str, float] = {}
        if cmp.tolerance_applied.absolute is not None:
            tol_dict["absolute"] = cmp.tolerance_applied.absolute
        if cmp.tolerance_applied.relative is not None:
            tol_dict["relative"] = cmp.tolerance_applied.relative
        report.append({
            "metric_name": cmp.metric_name,
            "expected": cmp.expected,
            "computed": cmp.computed,
            "delta": cmp.delta,
            "tolerance": tol_dict,
            "passed": cmp.passed,
        })
    return report
```

2. **`format_diff_table(report: list[dict[str, Any]]) -> str`**
   Takes the diff report list and formats it as a human-readable table string. Use fixed-width column alignment. Columns: METRIC (20 chars), EXPECTED (12), COMPUTED (12), DELTA (12), TOLERANCE (20), STATUS (6). Passing entries get "PASS", failing get "FAIL".

```python
def format_diff_table(report: list[dict[str, Any]]) -> str:
    """Format diff report as human-readable table string."""
    header = f"{'METRIC':<20} {'EXPECTED':>12} {'COMPUTED':>12} {'DELTA':>12} {'TOLERANCE':>20} {'STATUS':>6}"
    separator = "-" * len(header)
    lines = [header, separator]
    for entry in report:
        tol_str = ", ".join(f"{k}={v}" for k, v in entry["tolerance"].items())
        status = "PASS" if entry["passed"] else "FAIL"
        lines.append(
            f"{entry['metric_name']:<20} {entry['expected']:>12.6f} "
            f"{entry['computed']:>12.6f} {entry['delta']:>12.6f} "
            f"{tol_str:<20} {status:>6}"
        )
    return "\n".join(lines)
```

3. **`diff_report_to_json(report: list[dict[str, Any]]) -> str`**
   Serialize diff report to JSON string (using json.dumps with indent=2).

4. **`summarize_pass_rates(report: list[dict[str, Any]], registry: MetricRegistry) -> dict[str, Any]`**

```python
{

```

```python
@pytest.mark.golden
class TestL3DiffReport:

    def test_diff_report_structure(self):
        """Diff report produces correct structure with 6 keys per entry."""
        from stockvaluefinder.validation.schema import Tolerance
        from stockvaluefinder.validation.comparators import ComparisonResult

        comparisons = [
            ComparisonResult(
                metric_name="m_score",
                expected=-0.7421,
                computed=-0.74,
                delta=0.0021,
                tolerance_applied=Tolerance(absolute=0.05),
                passed=True,
            ),
            ComparisonResult(
                metric_name="f_score",
                expected=6.0,
                computed=7.0,
                delta=1.0,
                tolerance_applied=Tolerance(absolute=0),
                passed=False,
            ),
            ComparisonResult(
                metric_name="roic",
                expected=0.353242,
                computed=0.353242,
                delta=0.0,
                tolerance_applied=Tolerance(relative=0.01),
                passed=True,
            ),
        ]
        report = generate_diff_report(comparisons)

        assert len(report) == 3
        for entry in report:
            assert "metric_name" in entry
            assert "expected" in entry
            assert "computed" in entry
            assert "delta" in entry
            assert "tolerance" in entry
            assert "passed" in entry
        # Tolerance dict should only have non-None keys
        assert "absolute" in report[0]["tolerance"]
        assert "relative" not in report[0]["tolerance"]
        assert report[1]["passed"] is False
        assert report[2]["tolerance"] == {"relative": 0.01}

    def test_diff_report_json_serializable(self):
        """Diff report can be serialized to JSON and back."""
        import json
        from stockvaluefinder.validation.schema import Tolerance
        from stockvaluefinder.validation.comparators import ComparisonResult

        comparisons = [
            ComparisonResult("test1", 1.0, 1.01, 0.01, Tolerance(absolute=0.05), True),
            ComparisonResult("test2", 2.0, 2.5, 0.5, Tolerance(relative=0.1), False),
        ]
        report = generate_diff_report(comparisons)
        json_str = diff_report_to_json(report)
        parsed = json.loads(json_str)
        assert len(parsed) == 2
        assert parsed[0]["metric_name"] == "test1"
        assert parsed[1]["passed"] is False

    def test_format_diff_table(self):
        """Formatted table contains PASS/FAIL status and metric names."""
        from stockvaluefinder.validation.schema import Tolerance
        from stockvaluefinder.validation.comparators import ComparisonResult

        comparisons = [
            ComparisonResult("m_score", -0.74, -0.74, 0.0, Tolerance(absolute=0.05), True),
            ComparisonResult("f_score", 6.0, 7.0, 1.0, Tolerance(absolute=0), False),

        assert "FAIL" in table
```
