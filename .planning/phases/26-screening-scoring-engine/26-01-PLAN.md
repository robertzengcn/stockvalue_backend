def calculate_alpha_score(
    roic_wacc_score: float, capex_score: float, policy_score: float, moat_score: float,
    weights: tuple[float, float, float, float] = (0.40, 0.30, 0.20, 0.10),
) -> float:
    if len(weights) != 4:
        raise ValueError(...)
    if abs(sum(weights) - 1.0) > 0.01:
