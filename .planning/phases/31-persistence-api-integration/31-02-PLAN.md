          ```
          # Pledge risk analysis (graceful degradation per API-04)

          if request.include_pledge_risk:

                      snapshot = await data_service.get_equity_pledge_snapshot(ticker)
                      details = await data_service.get_equity_pledge_details(ticker)

                      pledge_analyzer = PledgeRiskAnalyzer()
                      pledge_risk_result = pledge_analyzer.analyze(
                          ticker=ticker,
                          snapshot=snapshot,
                          details=details,
                          financial_risk_level=risk_score.risk_level,
                          financial_red_flags=risk_score.red_flags,
                      )

                      # Persist pledge data (within same session/transaction)
                      pledge_snapshot_repo = PledgeSnapshotRepository(db)
                      pledge_detail_repo = PledgeDetailRepository(db)
                      if snapshot is not None:
                          snapshot_dict = snapshot.model_dump()
                          snapshot_dict.pop("data_quality", None)  # Remove non-DB field
                          snapshot_dict["source"] = snapshot.data_quality.source or "akshare"
                          snapshot_dict["latest_date"] = snapshot.latest_date
                          snapshot_dict["ticker"] = ticker
                          await pledge_snapshot_repo.upsert_by_ticker_date_source(
                              ticker=ticker,
                              latest_date=snapshot.latest_date,
                              source=snapshot_dict["source"],
                              data=snapshot_dict,
                          )
                      if details:
                          details_dicts = []
                          for d in details:
                              dd = d.model_dump()
                              dd["ticker"] = ticker
                              details_dicts.append(dd)
                          await pledge_detail_repo.replace_details_for_ticker(ticker, details_dicts)
                  else:
                      # API-05: HK tickers return unsupported result
                      pledge_risk_result = PledgeRiskAnalyzer().analyze(
                          ticker=ticker,
                          snapshot=None,
                          details=[],
                          financial_risk_level=risk_score.risk_level,
                          financial_red_flags=risk_score.red_flags,
                      )
              except Exception:
                  logger.warning(f"Pledge risk analysis failed for {ticker}", exc_info=True)
                  pledge_risk_result = None
          ```
